의존성: pyyaml(YAML 파서만), 표준 라이브러리.

polyprompt_simple.py
#!/usr/bin/env python3
# -*- coding: utf-8 -*-
"""
A minimal, pattern-free implementation for "prompt as Python object".
- No OOP design patterns (no Strategy/Composite/Builder).
- Single-file, dict-centric, pure functions.
- Base/Derived YAML → merge → render → (mock) infer.
Dependencies: pyyaml
"""

from __future__ import annotations
import json
from dataclasses import dataclass
from pathlib import Path
from typing import Any, Dict, List, Tuple
import yaml


# ----------------------------
# 0) Types (lightweight)
# ----------------------------
@dataclass
class PromptObject:
    """Materialized prompt object; just a dict wrapper."""
    name: str
    version: str
    template: str        # final joined string from sections
    variables: Dict[str, dict]   # {"var": {"type": "string|int|...","required":true,...}}
    constraints: List[str]
    examples: List[str]
    meta: Dict[str, Any] # anything else (system/instruction/output)


# ----------------------------
# 1) YAML Loaders (no classes)
# ----------------------------
def load_yaml_files(folder: str | Path) -> Dict[str, dict]:
    """Load all *.yaml under folder -> dict[name] = spec(dict)."""
    folder = Path(folder)
    specs: Dict[str, dict] = {}
    for p in folder.glob("*.yaml"):
        data = yaml.safe_load(p.read_text(encoding="utf-8"))
        if not isinstance(data, dict):
            raise ValueError(f"YAML must be a mapping: {p}")
        if "name" not in data or "type" not in data:
            raise ValueError(f"YAML must contain 'name' and 'type': {p}")
        specs[data["name"]] = data
    return specs


# ----------------------------
# 2) Merge (Base/Derived)
# ----------------------------
def merge_base_and_derived(all_specs: Dict[str, dict]) -> Dict[str, dict]:
    """
    Return fully-resolved specs: base kept as-is, derived merged with its base.
    Rules kept intentionally simple:
      - variables: shallow override if provided by derived
      - sections: 'system'/'instruction' replace if provided;
                  'constraints'/'examples' can use 'replace' OR 'append'/'remove'
      - output: shallow override
    """
    resolved: Dict[str, dict] = {}

    def resolve_one(name: str) -> dict:
        if name in resolved:
            return resolved[name]
        if name not in all_specs:
            raise KeyError(f"Unknown spec: {name}")

        spec = all_specs[name]
        t = spec.get("type")
        if t == "base":
            resolved[name] = spec
            return spec
        elif t == "derived":
            base_name = spec.get("base")
            if not base_name:
                raise ValueError(f"Derived must have 'base': {name}")
            base_spec = resolve_one(base_name)
            merged = _merge_two(base_spec, spec)
            resolved[name] = merged
            return merged
        else:
            raise ValueError(f"Unknown type: {t} in {name}")

    for n in list(all_specs.keys()):
        resolve_one(n)
    return resolved


def _merge_two(base: dict, derived: dict) -> dict:
    out = json.loads(json.dumps(base))  # deep copy via JSON (simple)
    out["name"] = derived["name"]
    if "description" in derived:
        out["description"] = derived["description"]

    # variables: shallow override/add
    if "variables" in derived and isinstance(derived["variables"], dict):
        if "variables" not in out or not isinstance(out["variables"], dict):
            out["variables"] = {}
        for k, v in derived["variables"].items():
            out["variables"][k] = v

    # sections
    out.setdefault("sections", {})
    sec = out["sections"]
    dsec = (derived.get("overrides", {}) or {}).get("sections", {})

    # replace simple strings
    if "system" in derived.get("sections", {}):
        sec["system"] = derived["sections"]["system"]
    if "instruction" in derived.get("sections", {}):
        sec["instruction"] = derived["sections"]["instruction"]

    # simple replacement from overrides
    if "system" in dsec:
        sec["system"] = dsec["system"]
    if "instruction" in dsec:
        sec["instruction"] = dsec["instruction"]

    # list fields
    for key in ("constraints", "examples"):
        # base default
        base_list = list(sec.get(key) or [])
        # direct replace from derived.sections
        if key in derived.get("sections", {}):
            maybe_list = derived["sections"][key]
            if isinstance(maybe_list, list):
                base_list = list(maybe_list)

        # overrides: replace/append/remove
        if key in dsec and isinstance(dsec[key], dict):
            patch = dsec[key]
            if "replace" in patch and patch["replace"] is not None:
                base_list = list(patch["replace"])
            if "append" in patch:
                base_list += list(patch["append"])
            if "remove" in patch:
                rm = set(patch["remove"])
                base_list = [x for x in base_list if x not in rm]

        sec[key] = base_list

    # output shallow override
    if "overrides" in derived and "output" in derived["overrides"]:
        out.setdefault("output", {})
        out["output"].update(derived["overrides"]["output"])
    if "output" in derived:
        out.setdefault("output", {})
        out["output"].update(derived["output"])

    return out


# ----------------------------
# 3) Compile → PromptObject
# ----------------------------
def compile_to_prompt(spec: dict, version: str = "v1") -> PromptObject:
    """Make the final template by concatenating sections; keep variables meta."""
    name = spec["name"]
    sections = (spec.get("sections") or {})
    system = sections.get("system", "")
    instruction = sections.get("instruction", "")
    constraints = sections.get("constraints", []) or []
    examples = sections.get("examples", []) or []

    blocks: List[str] = []
    if system.strip():
        blocks.append(f"[SYSTEM]\n{system}")
    if instruction.strip():
        blocks.append(instruction)
    if constraints:
        blocks.append("Constraints:\n" + "\n".join(f"- {c}" for c in constraints))
    if examples:
        blocks.append("Examples:\n" + "\n\n".join(examples))

    template = "\n\n".join(b for b in blocks if b.strip())
    variables = spec.get("variables") or {}
    meta = {
        "output": spec.get("output", {}),
        "description": spec.get("description", "")
    }

    return PromptObject(
        name=name,
        version=version,
        template=template,
        variables=variables,
        constraints=constraints,
        examples=examples,
        meta=meta
    )


# ----------------------------
# 4) Render & Infer
# ----------------------------
def _coerce_type(value: Any, t: str) -> Any:
    if t == "string":
        return str(value)
    if t == "int":
        return int(value)
    if t == "float":
        return float(value)
    if t == "bool":
        if isinstance(value, bool):
            return value
        s = str(value).lower()
        if s in ("true", "1", "yes", "y"):
            return True
        if s in ("false", "0", "no", "n"):
            return False
        raise ValueError(f"Cannot coerce to bool: {value}")
    raise ValueError(f"Unknown type: {t}")


def validate_and_cast_slots(variables: Dict[str, dict], slots: Dict[str, Any]) -> Dict[str, Any]:
    out: Dict[str, Any] = {}
    for k, spec in variables.items():
        required = bool(spec.get("required", True))
        if k not in slots:
            if required:
                raise ValueError(f"Missing required variable: {k}")
            else:
                out[k] = None
                continue
        v = slots[k]
        v = _coerce_type(v, spec.get("type", "string"))

        # simple bounds/length checks
        if isinstance(v, str):
            min_len = spec.get("min_length")
            max_len = spec.get("max_length")
            if min_len is not None and len(v) < int(min_len):
                raise ValueError(f"{k}: length<{min_len}")
            if max_len is not None and len(v) > int(max_len):
                raise ValueError(f"{k}: length>{max_len}")
        if isinstance(v, (int, float)):
            ge = spec.get("ge")
            le = spec.get("le")
            if ge is not None and float(v) < float(ge):
                raise ValueError(f"{k}: value<{ge}")
            if le is not None and float(v) > float(le):
                raise ValueError(f"{k}: value>{le}")

        out[k] = v

    # also allow extra slots (used in templates) but not declared -> pass-through
    for k, v in slots.items():
        if k not in out:
            out[k] = v
    return out


def render(po: PromptObject, **slots) -> str:
    typed = validate_and_cast_slots(po.variables, slots)
    try:
        return po.template.format(**typed)
    except KeyError as e:
        raise ValueError(f"Template variable not provided: {e}") from e


def infer_mock(po: PromptObject, **slots) -> str:
    """Mock inference: just returns rendered prompt with a delimiter."""
    prompt_text = render(po, **slots)
    return prompt_text + "\n\n[MOCK COMPLETION]"


# ----------------------------
# 5) Build-all helper
# ----------------------------
def build_prompts(folder: str | Path, version: str = "v1") -> Dict[str, PromptObject]:
    specs = load_yaml_files(folder)
    merged = merge_base_and_derived(specs)
    out: Dict[str, PromptObject] = {}
    for name, spec in merged.items():
        out[name] = compile_to_prompt(spec, version=version)
    return out


# ----------------------------
# 6) CLI (quick demo)
# ----------------------------
if __name__ == "__main__":
    import argparse
    ap = argparse.ArgumentParser(description="PolyPrompt minimal runner (pattern-free).")
    ap.add_argument("--prompts", type=str, default="prompts", help="Folder with *.yaml")
    ap.add_argument("--name", type=str, required=True, help="Prompt name to run")
    ap.add_argument("--slots", type=str, default="{}", help='JSON dict, e.g. {"text":"hi","length":40}')
    args = ap.parse_args()

    prompts = build_prompts(args.prompts)
    if args.name not in prompts:
        raise SystemExit(f"Unknown prompt: {args.name}. Available: {list(prompts)}")

    po = prompts[args.name]
    slots = json.loads(args.slots)
    print(infer_mock(po, **slots))

type: base
name: summarize.base
description: Base summarization prompt
variables:
  text:   { type: string, required: true, min_length: 1 }
  length: { type: int,    required: true, ge: 10, le: 500 }
sections:
  system: "You are a precise academic summarizer."
  instruction: "Summarize the following text in {length} words:\n{text}"
  constraints:
    - "Be factual and concise."
  examples:
    - "Input: 'Deep learning emerged...'; Output: 'Succinct summary...'"
output:
  format: free_text

  length: { type: int, required: true, ge: 50, le: 300 }

    system: "You are a precise legal summarizer. Cite statutes precisely."
